
fibonacci(int n) {
  if (n <= 1) {
    return n;
  }
  return fibonacci(n - 1) + fibonacci(n - 2);
}


 printFibonacciSeries(int n) {
  for (int i = 0; i < n; i++) {
    print(fibonacci(i));
  }
}

void main() {
  int terms = 10; 
  print("Fibonacci Series up to $terms terms:");
  printFibonacciSeries(terms);
}
