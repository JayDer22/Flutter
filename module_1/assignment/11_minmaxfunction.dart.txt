import'dart:io';

number(int num)
{
    int max = 0;
    int low = 9;
       
    
    while(num>0)
    {
        int rem=num%10;
        if(rem>max)
        {
            max=rem;
        }
        if(rem<low)
        {
           low=rem;  
        }
        num=num~/10;
    }
         print(max);
           print(low);
  
 
}
void main()
{
    print("Enter a number :");
    var num = int.parse(stdin.readLineSync().toString());
    number(num);
}


fibonacci(int n) {

  if (n <= 1) {

    return n;

  }

  return fibonacci(n - 1) + fibonacci(n - 2);

}





 printFibonacciSeries(int n) {

  for (int i = 0; i < n; i++) {

    print(fibonacci(i));

  }

}



void main() {

  int terms = 10; 

  print("Fibonacci Series up to $terms terms:");

  printFibonacciSeries(terms);

}
