import 'dart:io';
void main() 
{
    print("Enter your marks");
    var a = int.parse(stdin.readLineSync().toString());
    
    if(a>=90)
    {
        print("Grade A");
    }
    else if(a>=80)
    {
        print("Grade B");
    }
        else if(a>=70)
    {
        print("Grade C");
    }
        else if(a>=60)
    {
        print("Grade D");
    }
        else 
    {
        print("fail");
    }
}
