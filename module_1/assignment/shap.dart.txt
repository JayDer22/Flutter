import 'dart:io';
class Shape
{
    var t;
    void a()
    {
        print("Enter number for a:");
        t = int.parse(stdin.readLineSync().toString());
    }
}
class tringle extends Shape
{
    var r;
    void b()
    {
        print("Enter number for b:");
        r = int.parse(stdin.readLineSync().toString());
    }
}
class rectangle extends Shape
{
    void c(int r)
    {
        var tri = t*r;
        var rec = 0.5*t*r;
        print("Tringle : $tri");
        print("Tringle : $rec");
    }
}
void main() {
  tringle t1 = tringle();
  rectangle r1 = rectangle();

  t1.a();
  t1.b();
  r1.t = t1.t;
  r1.c(t1.r);
}
